{{- /* Responsive, compressed profile photo. The path comes from the "image" field in data/contact/details.json (a file in static/images/profile, mounted into assets in hugo.toml so Hugo can process it). The photo is cropped to a 3:4 portrait and served as WebP at several widths, so phones (near full-width photo) and desktop (280px) each download only what they need. Widths larger than the source can supply are skipped, so the photo is never upscaled. Params (dict): eager true for the above-the-fold photo (the LCP image) Usage: {{ partial "profile-image.html" (dict "eager" true) }} */ -}} {{- $contact := partial "contact-info.html" . -}} {{- with $contact.image | default "" | resources.Get -}} {{- $src := . -}} {{- $maxW := math.Min $src.Width (div (mul $src.Height 3) 4) | int -}} {{- $widths := slice -}} {{- range slice 280 360 440 560 720 -}} {{- if le . $maxW }}{{ $widths = $widths | append . }}{{ end -}} {{- end -}} {{- /* Small source: add its full width as the largest version */ -}} {{- if and (lt $maxW 720) (not (in $widths $maxW)) }}{{ $widths = $widths | append $maxW }}{{ end -}} {{- $srcset := slice -}} {{- $fallback := "" -}} {{- range $widths -}} {{- $img := $src.Fill (printf "%dx%d webp q78 Center" . (div (mul . 4) 3)) -}} {{- $srcset = $srcset | append (printf "%s %dw" $img.RelPermalink .) -}} {{- if or (not $fallback) (le . 440) }}{{ $fallback = $img }}{{ end -}} {{- end -}} {{- end -}}
